#1. Using simple dict
s = 'loveleetcode'
count = {}
for char in s:
    if char in count:
        count[char] += 1
    else:
        count[char] = 1

#2. from collection import defualtdict
count = defualtdict(int)
for char in s:
    count[char] += 1

#3. from collection import Counter
count = Counter(s)
